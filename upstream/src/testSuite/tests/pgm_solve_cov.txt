;*************************************************************
;** Coverage: program-based derivative and root solver.     **
;** covLoadPgm builds a labelled RPN program S (f(X)=X^2-4)  **
;** and loads it through fnLoadProgram; covDerivPgm and      **
;** covSolvePgm then differentiate and solve it through the  **
;** program (execProgram) branches of differentiate.c and   **
;** solve.c that the formula corpora cannot reach. Adds a    **
;** program to memory, so this runs late (serialize_cov,     **
;** which resets, stays last).                               **
;*************************************************************
In: FL_SPCRES=0 FL_CPXRES=0 SD=0 RMODE=0 IM=2compl SS=4 WS=64

; build and load program S: f(X) = X^2 - 4
Func: fnLoadPgmCov
In:  FARG=0
Out: EC=0

; first derivative of X^2-4 is 2X; at X=3 -> 6 (central difference is exact here)
Func: fnDerivPgmCov
In:  FARG=1 RX=Real:"3"
Out: EC=0 RX=Real:"6"

; second derivative of X^2-4 is 2 everywhere (central 2nd difference is h-exact)
Func: fnDerivPgmCov
In:  FARG=2 RX=Real:"3"
Out: EC=0 RX=Real:"2"

; solve f(X)=X^2-4=0 from guesses bracketing the positive root -> 2
Func: fnSolvePgmCov
In:  FARG=0 RY=Real:"1" RX=Real:"3"
Out: EC=0 RX=Real:"2"

; program-based integral of S (X^2-4) over [0,3] = [X^3/3 - 4X] = 9-12 = -3
Func: fnIntegratePgmCov
In:  FARG=0 RY=Real:"0" RX=Real:"3"
Out: EC=0 RX=Real:"-3"

; interactive selection (Integral f d) of the no-MVAR program T opens an empty MVAR menu, of program M (leading MVAR) a one-item menu; both write the menu list
; terminator into the byte _dynmenuConstructMVars allocates for it, which is the write that reboots the hardware when it lands outside (#500, #579)
Func: fnIntegrateErrCov
In:  FARG=2 RX=Real:"0"
Out: EC=0
Func: fnIntegrateErrCov
In:  FARG=3 RX=Real:"0"
Out: EC=0

; program summation Sum(n^2-4) for n=3..5 (from=3,to=5,step=1) = 5+12+21 = 38
Func: fnSumProdCov
In:  FARG=0 RZ=Real:"3" RY=Real:"5" RX=Real:"1"
Out: EC=0 RX=Real:"38"
; program product Prod(n^2-4) for n=3..5 = 5*12*21 = 1260
Func: fnSumProdCov
In:  FARG=1 RZ=Real:"3" RY=Real:"5" RX=Real:"1"
Out: EC=0 RX=Real:"1260"

; indexed (long-int) summation Sum(n^2) for n=1,3,5 (from=1,to=5,step=2) =
; 1+9+25 = 35 (distinct from/step verifies the stack-slot assignment)
Func: fnISumProdCov
In:  FARG=0 RZ=LonI:"1" RY=LonI:"5" RX=LonI:"2"
Out: EC=0 RX=LonI:"35"
; indexed product Prod(n^2) for n=1,3,5 = 1*9*25 = 225
Func: fnISumProdCov
In:  FARG=1 RZ=LonI:"1" RY=LonI:"5" RX=LonI:"2"
Out: EC=0 RX=LonI:"225"

; the MVAR page with no model selected: no VARMNU label, no formula, and currentSolverProgram at the 0xffff doFnReset leaves. The label index is bounded against
; numberOfLabels, so the page holds no variables. Program S is loaded by this point in the corpus, so the label block has program material after it
Func: fnMvarPageNoPgmCov
In:  FARG=0
Out: EC=0 RX=Real:"0"
