;*************************************************************
;** Coverage: equation derivative. fn1stDerivEq/fn2ndDerivEq **
;** differentiate the current formula f(X)=X^3 at the point   **
;** in X, evaluating the formula through parseEquation each    **
;** finite-difference step (differentiate.c + equation.c).     **
;** FARG selects the order; the eval point comes from X.       **
;*************************************************************
In: FL_SPCRES=0 FL_CPXRES=0 SD=0 RMODE=0 IM=2compl SS=4 WS=64

; first derivative of X^3 is 3*X^2 - exact at several points
Func: fnDerivEqCov
In:  FARG=1 RX=Real:"2"
Out: EC=0 RX=Real:"12"
Func: fnDerivEqCov
In:  FARG=1 RX=Real:"3"
Out: EC=0 RX=Real:"27"
Func: fnDerivEqCov
In:  FARG=1 RX=Real:"5"
Out: EC=0 RX=Real:"75"
; second derivative of X^3 is 6*X, and exact here too: the stencil is exact for a
; cubic and the step ladder settles where the cancellation is still invisible
Func: fnDerivEqCov
In:  FARG=2 RX=Real:"3"
Out: EC=0 RX=Real:"18"

; Directed result modes keep an exactly representable polynomial derivative exact.
Func: fnDerivEqCov
In:  FARG=1 RMODE=5 RX=Real:"2"
Out: EC=0 RX=Real:"12"
Func: fnDerivEqCov
In:  FARG=1 RMODE=6 RX=Real:"2"
Out: EC=0 RX=Real:"12"

; At x=0 the relative step uses its unit fallback. Floor mode from the preceding case remains selected; f'(0) of X^3
; is positive zero, so guard arithmetic must not introduce a negative sign.
Func: fnDerivEqCov
In:  FARG=1 RX=Real:"0"
Out: EC=0 RX=Real:"0"

; A finite real input is required. A complex scalar with a zero imaginary component is real-valued and accepted;
; non-real complex and matrix inputs retain their type and report INVALID_DATA_TYPE_FOR_OP (24).
Func: fnDerivEqCov
In:  FARG=1 RX=Cplx:"2i0"
Out: EC=0 RX=Real:"12"
Func: fnDerivEqCov
In:  FARG=1 RX=Cplx:"2i1"
Out: EC=24 RX=Cplx:"2i1"
Func: fnDerivEqCov
In:  FARG=1 RX=ReMa:"M1,2[2,3]"
Out: EC=24 RX=ReMa:"M1,2[2,3]"
Func: fnDerivEqCov
In:  FARG=1 RX=CxMa:"M1,1[2i0]"
Out: EC=24 RX=CxMa:"M1,1[2i0]"

; Infinite and NaN points have no finite neighbourhood. They return NaN without evaluating the formula.
Func: fnDerivEqCov
In:  FARG=1 RMODE=0 FL_SPCRES=1 RX=Real:"inf"
Out: EC=0 RX=Real:"NaN"
Func: fnDerivEqCov
In:  FARG=2 RMODE=0 FL_SPCRES=1 RX=Real:"NaN"
Out: EC=0 RX=Real:"NaN"

; program-derivative entry error paths (derivativeVariable): unknown label letter
; -> LABEL_NOT_FOUND (6), out-of-range parameter -> OUT_OF_RANGE (8)
Func: fnDerivErrCov
In:  FARG=0 RX=Real:"2"
Out: EC=6
Func: fnDerivErrCov
In:  FARG=1 RX=Real:"2"
Out: EC=8
