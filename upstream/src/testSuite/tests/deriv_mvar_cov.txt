;******************************************************************************************************************************
;** Coverage: derivative of a program that declares MVARs. covDerivMvarPgm builds and loads MD (f(x,p) = x^2 - p*x - 2 with  **
;** MVAR x and MVAR p) and differentiates it through fn1stDeriv / fn2ndDeriv. MD recalls its arguments from named storage,   **
;** so unlike the stack-reading program S of pgm_solve_cov it only moves when the differentiator stores each stencil point   **
;** in the variable it differentiates with respect to (differentiate.c). Without that the stencil samples are all equal and  **
;** the stencil coefficients, which sum to zero, cancel it exactly.                                                          **
;**                                                                                                                          **
;** Also asserts the variable choice among several MVARs - the current solver variable when the program declares it, else    **
;** the first declaration - and that sampling leaves both variables at the values they had on entry.                         **
;**                                                                                                                          **
;** Adds a program to memory and creates the named variables x, p and zzz, so this runs after programs.txt, next to          **
;** pgm_solve_cov, and before the resetting serialize_cov.                                                                   **
;******************************************************************************************************************************
In: FL_SPCRES=0 FL_CPXRES=0 SD=0 RMODE=0 IM=2compl SS=4 WS=64

; build and load M, then seed x = 5 and p = 0
Func: fnDerivMvarPgmCov
In:  FARG=0
Out: EC=0

; no solver variable selected: the first declaration, x, is the argument. d/dx = 2x - p, at x=2 with p=0 -> 4
Func: fnDerivMvarPgmCov
In:  FARG=5 RX=Real:"2"
Out: EC=0 RX=Real:"4"

; x selected explicitly: same derivative, at x=3 -> 6
Func: fnDerivMvarPgmCov
In:  FARG=1 RX=Real:"3"
Out: EC=0 RX=Real:"6"

; d2/dx2 = 2 everywhere
Func: fnDerivMvarPgmCov
In:  FARG=2 RX=Real:"3"
Out: EC=0 RX=Real:"2"

; p := 1 first: the result changes to 2x - 1 = 5, which only happens if the program reads p from named storage
Func: fnDerivMvarPgmCov
In:  FARG=3 RX=Real:"3"
Out: EC=0 RX=Real:"5"

; a solver variable MD does not declare falls back to the first declaration, x -> 2x - 1 = 5
Func: fnDerivMvarPgmCov
In:  FARG=8 RX=Real:"3"
Out: EC=0 RX=Real:"5"

; p selected: the caller's choice wins over the first declaration. d/dp = -x, with x = 5 -> -5
Func: fnDerivMvarPgmCov
In:  FARG=4 RX=Real:"1"
Out: EC=0 RX=Real:"-5"

; sampling perturbs the variables and must put them back: x is still its seeded 5 and p its assigned 1
Func: fnDerivMvarPgmCov
In:  FARG=6
Out: EC=0 RX=Real:"5"
Func: fnDerivMvarPgmCov
In:  FARG=7
Out: EC=0 RX=Real:"1"

; a long-integer input must come back a long integer: sampling stores a real34 into it, so the
; restore has to carry the whole register and not a real read back through real34.
Func: fnDerivMvarPgmCov
In:  FARG=9
Out: EC=0
Func: fnDerivMvarPgmCov
In:  FARG=1 RX=Real:"3"
Out: EC=0 RX=Real:"5"
Func: fnDerivMvarPgmCov
In:  FARG=6
Out: EC=0 RX=LonI:"5"
