;*************************************************************
;** Coverage: backup serializer round-trip. Save the whole  **
;** calculator state to backup.cfg and restore it, driving  **
;** both directions of saveRestoreBackup.c. This resets the  **
;** calculator, so it must be the last file in the list.     **
;**                                                          **
;** Then a regression on the same loader: the memory region  **
;** counts it reads from the file size the tables the        **
;** hexDump reader fills and every later walk of those       **
;** tables in freeList.c, so a count past the table ceiling  **
;** puts the writes and the walks off the end of a fixed     **
;** array. fnBackupBadRegionCount saves a genuine backup,    **
;** shadows one count line with 100000 (FARG 0 the free      **
;** regions, FARG 1 the allocated ones) and restores it. It  **
;** reports 1 only if both counts are inside their tables    **
;** afterwards, which they are when the loader refuses the   **
;** file; without the check the file's count is live and the **
;** case reports 0.                                          **
;*************************************************************
In: FL_SPCRES=0 FL_CPXRES=0 SD=0 RMODE=0 IM=2compl SS=4 WS=64

; decimal-valued backup fields preserve the Real34 range limits, signed zero,
; NaN and infinity independently of the raw register-memory image.
Func: fnBackupDecimalCov
In:  FARG=0
Out: EC=0 RX=Real:"1"

; malformed and inexact decimal text leaves the reset value unchanged
Func: fnBackupDecimalCov
In:  FARG=1
Out: EC=0 RX=Real:"1"
Func: fnBackupDecimalCov
In:  FARG=2
Out: EC=0 RX=Real:"1"
Func: fnBackupDecimalCov
In:  FARG=3
Out: EC=0 RX=Real:"1"
Func: fnBackupDecimalCov
In:  FARG=4
Out: EC=0 RX=Real:"1"

; the historical Real34 infinity spelling retains its sign
Func: fnBackupDecimalCov
In:  FARG=5
Out: EC=0 RX=Real:"1"

; seed a little varied state, then round-trip it through backup.cfg
Func: fnStore
In:  FARG=0 RX=Real:"1234.5678"
Out: EC=0
; the round-trip must be lossless: R00 stored above survives save+restore
Func: fnBackupRoundtrip
In:  FARG=0
Out: EC=0 R00=Real:"1234.5678"

; RM restored from backup.cfg sets the Real34 context too: RM 0 is selected between the save and the restore, and under RM 5, ceiling, 1/3 rounds up to ...334
Func: fnSetRoundingMode
In:  FARG=5 SD=0
Out: EC=0
Func: fnBackupRoundtrip
In:  FARG=1
Out: EC=0 RMODE=5
Func: fnDivide
In:  FL_ASLIFT=0 RY=LonI:"1" RX=LonI:"3"
Out: EC=0 RX=Real:"0.3333333333333333333333333333333334" ACC=34
Func: fnSetRoundingMode
In:  FARG=0
Out: EC=0

; an out-of-range free region count in the file must not become the live one
Func: fnBackupBadRegionCount
In:  FARG=0
Out: EC=0 RX=LonI:"1"

; and the same for the allocated region count
Func: fnBackupBadRegionCount
In:  FARG=1
Out: EC=0 RX=LonI:"1"
