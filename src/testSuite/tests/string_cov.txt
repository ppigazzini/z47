;*************************************************************
;** Coverage: ALPHA register / string operations.           **
;** Each case seeds the working string register (R00) fresh  **
;** so the mutated string or pushed result is deterministic  **
;** and asserted by value. FARG selects the string register. **
;*************************************************************
In: FL_SPCRES=0 FL_CPXRES=0 SD=0 RMODE=0 IM=2compl SS=4 WS=64

; clear the register to the empty string
Func: fnClearAlpha
In:  FARG=0 R00=Stri:"XX"
Out: EC=0 R00=Stri:""
; append the X string to the register
Func: fnXToAlpha
In:  FARG=0 R00=Stri:"AB" RX=Stri:"CD"
Out: EC=0 R00=Stri:"ABCD"
; length of the register content (glyph count) pushed to X
Func: fnAlphaLeng
In:  FARG=0 R00=Stri:"HELLO" RX=Real:"0"
Out: EC=0 RX=LonI:"5"
; ATOX: remove the first glyph; its code goes to X, the rest stays in R00
Func: fnAlphaToX
In:  FARG=0 R00=Stri:"HELLO" RX=Real:"0"
Out: EC=0 R00=Stri:"ELLO"
; position (0-based) of the X string inside the register
Func: fnAlphaPos
In:  FARG=0 R00=Stri:"HELLO" RX=Stri:"L"
Out: EC=0 RX=LonI:"2"
; rotate the register left by X glyphs
Func: fnAlphaRL
In:  FARG=0 R00=Stri:"HELLO" RX=Real:"1"
Out: EC=0 R00=Stri:"ELLOH"
; rotate the register right by X glyphs
Func: fnAlphaRR
In:  FARG=0 R00=Stri:"HELLO" RX=Real:"1"
Out: EC=0 R00=Stri:"OHELL"
; shift the register left by X glyphs (drop from the front)
Func: fnAlphaSL
In:  FARG=0 R00=Stri:"HELLO" RX=Real:"1"
Out: EC=0 R00=Stri:"ELLO"
; shift the register right by X glyphs (drop from the end)
Func: fnAlphaSR
In:  FARG=0 R00=Stri:"HELLO" RX=Real:"1"
Out: EC=0 R00=Stri:"HELL"
; append the integer part of X to the register
Func: fnAlphaIP
In:  FARG=0 R00=Stri:"HELLO" RX=Real:"3"
Out: EC=0 R00=Stri:"HELLO3"
